#!/usr/bin/env python3
"""Android Manifest Security Auditor - CLI tool for security vulnerability detection."""

from __future__ import annotations
import json, hashlib
from pathlib import Path
from typing import List, Tuple
from xml.etree import ElementTree as ET
import click

# Android namespace constant
A = "{http://schemas.android.com/apk/res/android}"

# Severity ordering for sorting output
ORDER = {"HIGH": 0, "MEDIUM": 1, "LOW": 2, "INFO": 3}

# System-level actions (used to detect receiver protection gaps)
SYSTEM_ACTIONS = {
    "android.intent.action.BOOT_COMPLETED",
    "android.intent.action.SCREEN_ON",
    "android.intent.action.SCREEN_OFF",
    "android.intent.action.TIME_SET",
    "android.intent.action.USER_PRESENT",
}

# Permission risk classification dictionary
RISKY_PERMS = {
    # HIGH severity - dangerous capabilities
    "BIND_ACCESSIBILITY_SERVICE": "HIGH",
    "PACKAGE_USAGE_STATS": "HIGH",
    "INSTALL_PACKAGES": "HIGH",
    "DELETE_PACKAGES": "HIGH",

    # MEDIUM severity - privacy/access concerns
    "MANAGE_EXTERNAL_STORAGE": "MEDIUM",
    "REQUEST_INSTALL_PACKAGES": "MEDIUM",
    "SYSTEM_ALERT_WINDOW": "MEDIUM",
    "READ_SMS": "MEDIUM",
    "RECEIVE_SMS": "MEDIUM",
    "READ_CALL_LOG": "MEDIUM",
    "PROCESS_OUTGOING_CALLS": "MEDIUM",

    # LOW severity - common but still sensitive
    "RECORD_AUDIO": "LOW",
    "READ_CONTACTS": "LOW",
    "ACCESS_FINE_LOCATION": "LOW",
    "QUERY_ALL_PACKAGES": "LOW",
    "WRITE_EXTERNAL_STORAGE": "LOW",
}


def sha256_file(path: Path) -> str:
    """Calculate SHA-256 hash of a file (chunked read for large files)."""
    h = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


def parse_manifest(path: Path) -> ET.Element:
    """Parse an AndroidManifest.xml file and return root element."""
    try:
        return ET.parse(str(path)).getroot()
    except Exception as e:
        raise click.ClickException(f"XML parse error: {e}")


def triage(root: ET.Element) -> List[Tuple[str, str, str]]:
    """
    Main security analysis engine.
    
    Returns a list of tuples: (severity, location, message)
    
    Checks performed:
    - Application-level flags (debuggable, allowBackup, cleartext)
    - Dangerous permission usage
    - Exported components without permission protection
    - Missing android:exported on targetSdk 31+
    """
    # Get target SDK version
    sdk = root.find("uses-sdk")
    target = int(sdk.get(f"{A}targetSdkVersion", "0")) if sdk is not None else 0
    
    # Get application element
    app = root.find("application")
    out: List[Tuple[str, str, str]] = []
    
    # Helper to add findings
    def add(sev: str, where: str, msg: str) -> None:
        out.append((sev, where, msg))
    
    # === APPLICATION-LEVEL CHECKS ===
    if app is not None:
        # Debuggable flag in production builds
        if app.get(f"{A}debuggable") == "true":
            add("HIGH", "application", "debuggable=true")
        
        # Backup enabled allows data exfiltration
        if app.get(f"{A}allowBackup", "true") == "true":
            add("MEDIUM", "application", "allowBackup=true")
        
        # Cleartext HTTP traffic (unencrypted)
        ct = app.get(f"{A}usesCleartextTraffic")
        if ct == "true" or (ct is None and 0 < target < 28):
            add("MEDIUM", "application", "cleartext HTTP permitted")
    
    # === PERMISSION CHECKS ===
    for p in root.findall("uses-permission"):
        name = p.get(f"{A}name", "")
        short = name.rsplit(".", 1)[-1] if "." in name else name
        if short in RISKY_PERMS:
            add(RISKY_PERMS[short], "uses-permission", short)
    
    # === COMPONENT EXPORT CHECKS ===
    if app is not None:
        for el in app.iter():
            # Only check component elements
            if el.tag not in ("activity", "activity-alias", "service", "receiver", "provider"):
                continue
            
            name = el.get(f"{A}name", "?")
            filters = el.findall("intent-filter")
            attr = el.get(f"{A}exported")
            
            # Android 31+ requires explicit exported attribute
            if target >= 31 and attr is None and filters:
                add("HIGH", f"{el.tag} {name}", "android:exported missing on targetSdk 31+")
            
            # Determine if component is actually exported
            exported = (attr == "true") if attr is not None else bool(filters)
            if not exported:
                continue  # Not exported, skip further checks
            
            # Check for permission protection
            actions = sorted({a.get(f"{A}name", "") for f in filters for a in f.findall("action")})
            perm = el.get(f"{A}permission")
            
            # Providers can have read/write specific permissions
            if el.tag == "provider":
                perm = perm or el.get(f"{A}readPermission") or el.get(f"{A}writePermission")
            
            # Already permission-protected? Skip
            if perm:
                continue
            
            # Classify the export risk
            is_launcher = "android.intent.action.MAIN" in actions
            sys_only = bool(actions) and all(a in SYSTEM_ACTIONS for a in actions)
            grant = el.get(f"{A}grantUriPermissions") == "true"
            
            if is_launcher:
                sev, msg = "INFO", "launcher entry"
            elif el.tag == "provider":
                sev, msg = "HIGH", f"exported provider, no permission{' + grantUriPermissions' if grant else ''}"
            elif el.tag == "service":
                sev, msg = "HIGH", "exported service, no permission"
            elif el.tag == "receiver" and sys_only:
                sev, msg = "LOW", "exported receiver for system-only actions"
            elif el.tag == "receiver":
                sev, msg = "MEDIUM", "exported receiver, no permission"
            else:
                sev, msg = "INFO", f"exported {el.tag} {name}"
            
            out.append((sev, f"{el.tag} {name}", msg))
    
    # Sort by severity order
    return sorted(out, key=lambda x: ORDER[x[0]])


def render_html(root, issues, sha) -> str:
    """
    Generate HTML report with color-coded severity rows.
    
    Designed to be viewed in terminal via:
    w3m -T text/html -dump report.html
    """
    package = root.get("package", "?")
    
    # Count severities for summary badges
    high_count = sum(1 for s,_,_ in issues if s=="HIGH")
    med_count = sum(1 for s,_,_ in issues if s=="MEDIUM")
    low_count = sum(1 for s,_,_ in issues if s=="LOW")
    info_count = sum(1 for s,_,_ in issues if s=="INFO")
    
    # Build table rows
    rows = "\n".join(
        f"<tr><td class='{sev}'><b>{sev}</b></td><td>{loc}</td><td>{msg}</td></tr>"
        for sev, loc, msg in issues
    )
    
    return f'''<!doctype html>
<html>
<head><meta charset="utf-8">
<title>Audit Report: {package}</title>
<style>
body {{ font-family: sans-serif; margin: 2rem; }}
h1 {{ color: #333; border-bottom: 2px solid #6d4aff; padding-bottom: 0.5rem; }}
table {{ border-collapse: collapse; width: 100%; }}
td, th {{ border: 1px solid #ccc; padding: 0.5rem; }}
th {{ background: #f0f0f0; }}
.HIGH {{ background: #ffe1e1; color: #c00; }}
.MEDIUM {{ background: #fff0cc; color: #a60; }}
.LOW {{ background: #eef4ff; color: #36c; }}
.INFO {{ background: #f5f5f5; color: #666; }}
.summary {{ display: flex; gap: 1rem; margin: 1rem 0; }}
.count {{ padding: 1rem; border-radius: 4px; text-align: center; min-width: 80px; }}
.count.high {{ background: #ffe1e1; }}
.count.med {{ background: #fff0cc; }}
.count.low {{ background: #eef4ff; }}
.count.info {{ background: #f5f5f5; }}
</style>
</head>
<body>
<h1>Android Manifest Audit Report</h1>
<div class="meta"><b>Package:</b> {package}<br><b>SHA-256:</b> <code>{sha}</code></div>
<div class="summary">
<div class="count high"><b>HIGH</b><br>{high_count}</div>
<div class="count med"><b>MEDIUM</b><br>{med_count}</div>
<div class="count low"><b>LOW</b><br>{low_count}</div>
<div class="count info"><b>INFO</b><br>{info_count}</div>
</div>
<table>
<tr><th>Severity</th><th>Location</th><th>Issue</th></tr>
{rows if rows else "<tr><td colspan='3'><i>No issues found</i></td></tr>"}
</table>
</body></html>'''


@click.command()
@click.argument("manifest", type=click.Path(exists=True, path_type=Path))
@click.option("--json", "as_json", is_flag=True, help="Emit JSON output")
@click.option("--html", "as_html", is_flag=True, help="Generate HTML report")
@click.option("--out", type=click.Path(path_type=Path), default=None, help="Write to file")
def cli(manifest: Path, as_json: bool, as_html: bool, out: Path | None):
    """
    Android Manifest Security Auditor
    
    Scan an AndroidManifest.xml file for security vulnerabilities.
    
    Examples:
        audit_manifest manifest.xml                  # Text output
        audit_manifest manifest.xml --json           # JSON output
        audit_manifest manifest.xml --html -o r.html # HTML report
    """
    # Parse manifest
    root = parse_manifest(manifest)
    
    # Run security triage
    issues = triage(root)
    
    # Count severities
    counts = {"HIGH": 0, "MEDIUM": 0, "LOW": 0, "INFO": 0}
    for sev, _, _ in issues:
        counts[sev] += 1
    
    # Calculate file hash
    sha = sha256_file(manifest)
    
    # Generate output based on format selection
    if as_html:
        text = render_html(root, issues, sha)
    elif as_json:
        payload = {
            "package": root.get("package", "?"),
            "sha256": sha,
            "issues": [{"severity": s, "location": l, "message": m} for s, l, m in issues],
            "counts": counts
        }
        text = json.dumps(payload, indent=2)
    else:
        lines = [f"[{sev}] {loc}: {msg}" for sev, loc, msg in issues]
        lines.append("")
        lines.append(f"Severity counts: HIGH={counts['HIGH']} MEDIUM={counts['MEDIUM']} LOW={counts['LOW']} INFO={counts['INFO']}")
        text = "\n".join(lines)
    
    # Write or print output
    if out:
        out.write_text(text)
        click.echo(f"Report written to {out}")
    else:
        click.echo(text)


if __name__ == "__main__":
    cli()

"""Unit tests for Android Manifest Security Auditor."""

from pathlib import Path
import sys

# Ensure src directory is in path
sys.path.insert(0, str(Path(__file__).parent.parent / "src"))

from audit_manifest import parse_manifest, triage

FIXTURES = Path(__file__).parent / "fixtures"


def test_safe_has_no_highs():
    """Verify safe manifest produces zero HIGH severity issues."""
    root = parse_manifest(FIXTURES / "AndroidManifest_safe.xml")
    issues = triage(root)
    severities = [sev for sev, _, _ in issues]
    assert "HIGH" not in severities


def test_risky_flags_issues():
    """Verify risky manifest flags both HIGH and MEDIUM issues."""
    root = parse_manifest(FIXTURES / "AndroidManifest_risky.xml")
    issues = triage(root)
    severities = [sev for sev, _, _ in issues]
    
    # Should have at least one HIGH issue
    assert "HIGH" in severities
    
    # Should have at least one MEDIUM issue
    assert "MEDIUM" in severities
